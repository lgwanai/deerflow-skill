    ```python
    """Tool registry logging and visibility.

    This module provides functions to log available tools at startup,
    providing visibility into deer-flow's tool ecosystem.

    The skill does NOT load tools directly - deerflow-harness handles
    all tool loading. This module only provides logging visibility.

    Usage:
        from lib.tools import log_available_tools

        tools = get_available_tools()  # from deerflow
        log_available_tools(tools)
    """
    import sys
    from typing import TYPE_CHECKING

    if TYPE_CHECKING:
        from langchain_core.tools import BaseTool


    BUILTIN_TOOL_NAMES = ["bash", "read", "write", "str_replace"]


    def get_unique_tool_names(tools: list["BaseTool"]) -> list[str]:
        """Get unique tool names from a list of tools (TOOL-03).

        Deduplicates tools by name, preserving first occurrence order.
        This helper is for testing - deerflow-harness handles dedup internally.

        Args:
            tools: List of BaseTool objects.

        Returns:
            List of unique tool names.
        """
        seen = set()
        names = []
        for tool in tools:
            if tool.name not in seen:
                seen.add(tool.name)
                names.append(tool.name)
        return names


    def log_available_tools(tools: list["BaseTool"]) -> None:
        """Log available tools at startup (TOOL-01).

        Prints tool names to stderr for visibility:
        - Built-in tools first
        - Other tools sorted alphabetically

        Args:
            tools: List of BaseTool objects from deerflow.tools.get_available_tools().
        """
        # Separate built-in and other tools
        builtin = [t for t in tools if t.name in BUILTIN_TOOL_NAMES]
        others = [t for t in tools if t.name not in BUILTIN_TOOL_NAMES]

        # Log header
        print(f"\n[Tools loaded: {len(tools)}]", file=sys.stderr, flush=True)

        # Log built-in tools first
        if builtin:
            print("[Built-in tools:]", file=sys.stderr, flush=True)
            for tool in sorted(builtin, key=lambda t: t.name):
                print(f"  - {tool.name}", file=sys.stderr, flush=True)

        # Log other tools
        if others:
            print("[Other tools:]", file=sys.stderr, flush=True)
            for tool in sorted(others, key=lambda t: t.name):
                print(f"  - {tool.name}", file=sys.stderr, flush=True)
    ```
